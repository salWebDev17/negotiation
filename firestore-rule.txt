rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Anyone visiting the site can read the video list.
    // Only visaldy1@gmail.com (signed in) can add, edit, or delete videos.
    match /videos/{videoId} {
      allow read: if true;
      allow write: if request.auth != null
                   && request.auth.token.email == "visaldy1@gmail.com";
    }

    // Any visitor's browser can create ONE visit record (that's how the
    // site logs a visit). Only the admin can read/delete the visit log.
    match /visits/{visitId} {
      allow create: if true;
      allow read, update, delete: if request.auth != null
                   && request.auth.token.email == "visaldy1@gmail.com";
    }
  }
}
